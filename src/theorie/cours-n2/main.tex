% Sauvegarde de la commande \part originale de Beamer
\let\beamer@originalpart\part

% Redéfinition de \part :
%   \part{Titre long pour partpage}              → TOC = Titre long
%   \part[Titre court pour TOC]{Titre long}      → TOC = Titre court
\renewcommand{\part}[2][]{%
  \def\beamer@parttocname{#1}%
  \ifx\beamer@parttocname\empty
    % Pas d'argument optionnel → on utilise le titre principal pour la TOC
    \def\beamer@parttocname{#2}%
  \fi
  \beamer@originalpart{#2}%
}

\AtBeginPart{%
  % Enregistre le numéro de partie, le nom et la page dans le fichier .parttoc
  \addtocontents{parttoc}{\protect\beamer@partintoc{\the\c@part}{\beamer@parttocname}{\the\c@page}}%
  % This adds a slide with the Part title automatically when you use \part{}
  \begin{frame}
    \centering
    % On recrée le bloc bleu manuellement pour garder le style
    \begin{beamercolorbox}[sep=16pt,center,shadow=true,rounded=true]{part title}
      \usebeamerfont{part title}
      \insertpart\par
    \end{beamercolorbox}
  \end{frame}
}

% Création de la commande pour afficher ce sommaire
\newcommand{\parttableofcontents}{\@starttoc{parttoc}}

% Utilisation de \newcommand car Beamer ne possède pas de beamer@partintoc par défaut
\newcommand{\beamer@partintoc}[3]{%
  \ifnum#1>0\relax
    \leavevmode
    \begingroup
        % On synchronise le compteur pour le bullet
        \c@section=#1\relax
        % On définit les variables attendues par le template du thème
        \def\inserttocsectionnumber{#1}%
        \def\inserttocsection{\hyperlink{page.#3}{#2}}% Définit le texte cliquable
        \def\inserttocbookmark{}%
        \def\inserttocfullsection{\hyperlink{page.#3}{#2}}%
        \def\inserttocfulltag{}%
        % On appelle le template du thème (qui utilisera \inserttocsection)
        \usebeamertemplate{section in toc}%
    \endgroup
  \fi
}
