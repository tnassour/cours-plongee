\section{Les Tables de plongée}

\begin{frame}{Introduction aux tables MN90 - FFESSM}
    \begin{itemize}
        \item \textbf{Origine :} Établies par la Marine Nationale en 1990 et adoptées par la FFESSM pour la plongée récréative.
        \item \textbf{Fondement scientifique :} Basées sur une modélisation du corps humain et des résultats statistiques d'expérimentations, à l'instar des tables de Haldane.
        \item \textbf{Modélisation des tissus :}
        \begin{itemize}
            \item Le corps est représenté par \textbf{12 compartiments fictifs}.
            \item Chaque compartiment a une capacité d'absorption ou d'élimination de l'azote équivalente aux tissus réels qu'il représente.
        \end{itemize}
        \item \textbf{Objectif :} Prévenir l'Accident de Désaturation (ADD) causé par la présence de bulles d'azote dans le corps.
    \end{itemize}
\end{frame}

\begin{frame}{Calcul avec les tables : Les règles de base}
    \begin{itemize}
        \item \textbf{Le modèle "Carré" :} Les tables MN90 considèrent que le plongeur reste à sa \textcolor{ffessmred}{\textbf{profondeur maximale}} pendant toute la durée de l'immersion.
        \item \textbf{Détermination de la durée :}
        \begin{itemize}
            \item \textbf{Départ :} Début de la descente.
            \item \textbf{Fin :} Moment où le plongeur quitte le fond pour entamer sa remontée (vitesse de 15 à 17 m/min).
        \end{itemize}
        \item \textbf{Sécurité (Lecture "en haut") :} Si les paramètres réels ne figurent pas exactement dans la table :
        \begin{itemize}
            \item On choisit la profondeur immédiatement \textbf{supérieure}.
            \item On choisit la durée immédiatement \textbf{supérieure}.
        \end{itemize}
        \item \textbf{Cas particulier :} Si la remontée est lente ($< 15$ m/min), le temps jusqu'au premier palier est ajouté à la durée de plongée.
    \end{itemize}
\end{frame}

\begin{frame}{La chaîne de calcul : GPS et Majoration}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{1. Sortie de la $1^{ère}$ plongée}
            La table donne un \textbf{GPS} (Groupe de Plongée Successive). Cette lettre représente l'azote résiduel dans le sang.
        \end{block}
        
        \begin{block}{2. Intervalle de surface}
            Selon le temps passé hors de l'eau, le \textbf{taux d'azote} diminue. On trouve ce nouveau taux dans le \textbf{Tableau 1}.
        \end{block}

        \column{0.48\textwidth}
        \begin{block}{3. Calcul de la $2^{nde}$ plongée}
            On utilise le \textbf{Tableau 2} pour trouver la \textbf{Majoration} (en minutes).
        \end{block}
        
        \begin{alertblock}{Règle d'or}
            \centering
            \textbf{Durée fictive =} \\
            Durée réelle + Majoration
        \end{alertblock}
    \end{columns}
    \vspace{0.3cm}
    \textit{Note : C'est cette \textbf{durée fictive} que l'on cherche dans la table pour déterminer les nouveaux paliers.}
\end{frame}

\begin{frame}{Avantages et Limites des Tables}
    \begin{columns}
        \begin{column}{0.4\textwidth}
            \textbf{Avantages :}
            \begin{itemize}
                \item Fiabilité (pas de pile).
                \item Cadre rigoureux.
                \item Idéal pour la planification.
            \end{itemize}
        \end{column}
        \begin{column}{0.4\textwidth}
            \textbf{Limites :}
            \begin{itemize}
                \item Très pénalisantes (profil carré).
                \item Erreur humaine de lecture possible.
                \item Pas de calcul en temps réel si le profil change.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
