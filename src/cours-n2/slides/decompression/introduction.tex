\section{Introduction}

\begin{frame}{Pourquoi calculer sa décompression ?}
    \begin{itemize}
        \item \textbf{Objectif :} Éviter l'Accident de Désaturation (ADD).
        \item \textbf{Principe :} Gérer l'élimination de l'azote ($N_2$) accumulé sous pression (Loi de Henry).
        \item \textbf{Cadre réglementaire (Code du Sport) :} 
        \begin{itemize}
            \item Obligatoire pour toute plongée en autonomie.
            \item Obligatoire pour toute plongée au-delà de 20 mètres.
        \end{itemize}
    \end{itemize}
    \begin{block}{Le rôle du plongeur N2}
        Devenir responsable de sa propre sécurité et de celle de ses équipiers en gérant ses paramètres de remontée.
    \end{block}
\end{frame}
