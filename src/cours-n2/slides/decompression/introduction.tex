% --- INTRODUCTION ET OBJECTIFS ---
\begin{frame}{Pourquoi calculer sa décompression ?}
    \begin{itemize}
        \item \textbf{Objectif vital :} Éviter l'Accident de Désaturation (ADD).
        \item \textbf{Le défi :} L'azote ($N_2$) accumulé durant la plongée doit être évacué de manière contrôlée lors de la remontée.
    \end{itemize}
    
    \begin{block}{Le rôle du plongeur N2}
        Devenir responsable de sa propre sécurité et de celle de ses équipiers en gérant activement ses paramètres de remontée pour rester dans les clous de la désaturation.
    \end{block}
\end{frame}

% --- LE CADRE LÉGAL (ART. A. 322-80) ---
\begin{frame}{Cadre réglementaire (Code du Sport)}
    Selon l'\textbf{Art. A. 322-80}, le contrôle des paramètres est obligatoire en milieu naturel :
    \begin{itemize}
        \item \textbf{Équipements obligatoires :} \textcolor{ffessmred}{Tout plongeur doit pouvoir contrôler les caractéristiques personnelles de sa plongée et de sa remontée (\textbf{ordinateur} ou \textbf{tables/timer}).}
        \item \textbf{Publics concernés :} 
        \begin{itemize}
            \item Chaque plongeur en \textbf{autonomie}.
            \item Chaque plongeur \textbf{encadré au-delà de 20 mètres}.
        \end{itemize}
        \item \textbf{Sécurité :} Obligation d'un système d'alimentation de secours (\textbf{octopus}) sans partage d'embout.
    \end{itemize}
\end{frame}

% --- LA LOI DE HENRY (PHYSIQUE) ---
\begin{frame}{La Loi de Henry (1803)}
    \textbf{Loi de dissolution des gaz dans les liquides :}
    
    \begin{itemize}
        \item \textbf{Principe :} À température constante et à saturation, la quantité de gaz dissous dans un liquide est proportionnelle à la pression partielle qu'exerce ce gaz sur le liquide.
    \end{itemize}

    \begin{equation*}
        V = k \cdot P_p
    \end{equation*}
    \centering \textit{\small ($V$ : Volume de gaz dissous, $k$ : coefficient de solubilité, $P_p$ : Pression partielle du gaz)}

    \begin{itemize}
        \item \textbf{Mécanisme :} 
        \begin{itemize}
            \item Le gaz se dissout dans le liquide du fait de l'agitation des molécules.
            \item \textbf{Pression :} Si elle augmente, l'agitation moléculaire augmente et une plus grande quantité de gaz se dissout.
            \item \textbf{Température :} La quantité de gaz dissous augmente également quand la température baisse.
        \end{itemize}
    \end{itemize}
\end{frame}

% --- CONSÉQUENCES POUR LE PLONGEUR ---
\begin{frame}{Conséquences pour le plongeur}
    \begin{columns}[c]
        \column{0.5\textwidth}
        \begin{center}
            \includegraphics[width=\columnwidth]{images/azote_et_plongee}
        \end{center}

        \column{0.5\textwidth}
        \only<1>{
            \begin{center}
                \begin{enumerate}
                    \item \textbf{À la descente :} La pression augmente, notre corps stocke de l'azote ($N_2$) dans les tissus.
                    \item \textbf{À la remontée :} La pression diminue, l'azote doit être évacué par les poumons par expiration.
                \end{enumerate}
            \end{center}
        }
        \only<2->{
            \begin{center}
                \textbf{Danger :} Si la décompression est trop rapide, l'azote sort des tissus sous forme de \textbf{bulles} avant d'atteindre les poumons.

                \begin{alertblock}{Risque majeur}
                    Une remontée trop rapide ou le non-respect des paliers provoque l'Accident de Désaturation (ADD).
                \end{alertblock}
            \end{center}
        }
    \end{columns}
\end{frame}

\begin{frame}{L'Accident de Décompression (ADD)}
    \begin{columns}[c]
        \column{0.5\textwidth}
        \textbf{Cause principal :}
        \begin{itemize}
            \item Présence de \textbf{bulles d'azote} dans le corps.
        \end{itemize}

        \vspace{0.5cm}
        
        \textbf{Symptômes :}
        \begin{itemize}
            \item Paralysie des membres.
            \item Difficultés pour uriner.
            \item Troubles de la vision.
            \item Troubles de la parole.
            \item Troubles de l'équilibre.
        \end{itemize}

        \column{0.5\textwidth}
        \begin{center}
            \includegraphics[width=\columnwidth]{images/risques_decompression}
        \end{center}
    \end{columns}
\end{frame}

% --- LES OUTILS DE CALCUL ---
\begin{frame}{Comment éviter ou réduire le risque de l'ADD ?}
    \begin{itemize}
        \item \textbf{Les Tables (ex: MN92) :} Procédure forfaitaire de remontée.
        \item \textbf{L'ordinateur de plongée :} Calcule en temps réel la saturation selon des modèles mathématiques intégrant la loi de Henry.
    \end{itemize}
    
    \begin{block}{Le Manomètre}
        \textbf{Le Manomètre :} Indispensable pour vérifier que l'on possède assez d'air pour effectuer ses paliers.
        
        "Chaque bouteille est munie d'un manomètre permettant d'indiquer la pression au cours de la plongée" (Art. A. 322-80).
    \end{block}
\end{frame}
